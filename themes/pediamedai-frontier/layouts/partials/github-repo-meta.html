{{- /* GitHub repo metadata. Live API → fallback to data/repos.yaml. Hugo caches resources.GetRemote responses (default 30 min) so repeated builds don't re-fetch. The fallback covers offline / rate-limited builds. Args: repo (required) e.g. "PediaMedAI/AggPose" Output: bare spans separated by aria-hidden middots; styling is inherited from the surrounding .card-meta wrapper. */ -}} {{- $url := printf "https://api.github.com/repos/%s" .repo -}} {{- $data := index $.Site.Data.repos .repo -}} {{- with try (resources.GetRemote $url) -}} {{- with .Value -}} {{- with (try (transform.Unmarshal .)).Value -}}{{- $data = . -}}{{- end -}} {{- end -}} {{- end -}} {{- with $data -}} {{- $stars := .stargazers_count | default 0 -}} {{- $lang := .language | default "" -}} {{- $pushed := .pushed_at | default "" -}} ★ {{ $stars }} {{- with $lang }} · {{ . }}{{ end -}} {{- with $pushed -}} {{- $t := time . -}} · Updated {{ $t.Format "Jan 2, 2006" }} {{- end -}} {{- end -}}
