{{- /* GitHub repo metadata — fetched live from the GitHub API at build time, with a fallback to data/repos.yaml when the live call fails (network down, rate-limited, transient 5xx). The partial renders nothing only if BOTH paths fail (e.g. the repo isn't in the YAML and the API is unreachable). Hugo caches resources.GetRemote responses (default 30 min) so multiple builds in quick succession don't re-fetch. Args: repo (required) e.g. "PediaMedAI/AggPose" Output: a sequence of plain spans separated by aria-hidden middot spans. Styling is inherited from .card-meta on the wrapping element. */ -}} {{- $repo := .repo -}} {{- /* —— Live fetch —— */ -}} {{- $url := printf "https://api.github.com/repos/%s" $repo -}} {{- $r := try (resources.GetRemote $url) -}} {{- $data := dict -}} {{- with $r.Value -}} {{- $parsed := try (transform.Unmarshal .) -}} {{- with $parsed.Value -}} {{- $data = . -}} {{- end -}} {{- end -}} {{- /* —— Fallback to snapshot if live fetch yielded nothing —— */ -}} {{- if not $data -}} {{- with index $.Site.Data.repos $repo -}} {{- $data = . -}} {{- end -}} {{- end -}} {{- with $data -}} {{- $stars := .stargazers_count | default 0 -}} {{- $lang := .language | default "" -}} {{- $pushed := .pushed_at | default "" -}} ★ {{ $stars }} {{- with $lang }} · {{ . }}{{ end -}} {{- if $pushed -}} {{- $t := time $pushed -}} · Updated {{ $t.Format "Jan 2, 2006" }} {{- end -}} {{- end -}}
